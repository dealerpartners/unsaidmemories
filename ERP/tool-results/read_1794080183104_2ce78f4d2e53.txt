     8→export interface BillItem {
     9→  id: string;
    10→  vendor: string;
    11→  source: 'Auto-scanned' | 'Email forwarded' | 'Manual' | 'Upload';
    12→  billNo: string;
    13→  billDate: string;
    14→  category: string;
    15→  className?: string;
    16→  location?: string;
    17→  memo: string;
    18→  dueDate: string;
    19→  amount: number;
    20→  attachmentsCount: number;
    21→  status: 'For review' | 'Unpaid' | 'Paid';
    22→}
    23→
    24→/* ═══ Store → screen adapters ═══
    25→   The screen keeps its approved layout and local row shape; every row is now
    26→   derived from the shared platform store (§1.1 single source of truth). */
    27→
    28→/* The store uses ISO dates; the approved screen shows MM/DD/YYYY */
    29→const toDisplayDate = (iso: string): string => {
    30→  const m = /^(\d{4})-(\d{2})-(\d{2})$/.exec((iso || '').trim());
    31→  return m ? `${m[2]}/${m[3]}/${m[1]}` : iso || '—';
    32→};
    33→
    34→/* The create form shows MM/DD/YYYY; the store stores ISO */
    35→const toIsoDate = (display: string): string => {
    36→  const m = /^(\d{2})\/(\d{2})\/(\d{4})$/.exec((display || '').trim());
    37→  return m ? `${m[3]}-${m[1]}-${m[2]}` : display;
    38→};
    39→
    40→/* Source badges derive from tags (set on create) and the activity log */
    41→const SOURCE_FROM_TAG: Record<string, BillItem['source']> = {
    42→  'source:upload': 'Upload',
    43→  'source:email': 'Email forwarded',
    44→  'source:scan': 'Auto-scanned',
    45→};
    46→
    47→/* Store DocumentStatus → the screen's three tabs: For review / Unpaid / Paid.
    48→   Drafts land in 'For review' (upload → review → approve → paid funnel);
    49→   Approved/Overdue/etc. are payable, so they land in 'Unpaid'. */
    50→const toTabStatus = (s: DocumentStatus): BillItem['status'] =>
    51→  s === 'ForReview' || s === 'Draft'
    52→    ? 'For review'
    53→    : s === 'Paid' || s === 'Deposited'
    54→    ? 'Paid'
    55→    : 'Unpaid';
    56→
    57→/* Next bill number per the platform numbering scheme */
    58→const nextBillNumber = () =>
    59→  `BILL-${usePlatformStore.getState().documents.filter((d) => d.type === 'Bill').length + 2001}`;
    60→
    61→/* Store Document → the screen's local row type */
    62→function toBillItem(doc: Document, vendorName: string): BillItem {
    63→  const taggedSource = doc.tags.map((t) => SOURCE_FROM_TAG[t]).find(Boolean);
    64→  const activityText = doc.activity.map((a) => a.text).join(' ').toLowerCase();
    65→  const source: BillItem['source'] =
    66→    taggedSource ??
    67→    (activityText.includes('email')
    68→      ? 'Email forwarded'
    69→      : activityText.includes('upload') || activityText.includes('scan')
    70→      ? 'Auto-scanned'
    71→      : 'Manual');
    72→  return {
    73→    id: doc.id,
    74→    vendor: vendorName || '—',
    75→    source,
    76→    billNo: doc.number,
    77→    billDate: toDisplayDate(doc.date),
    78→    category: doc.lines[0]?.description || 'Uncategorized',
    79→    className: doc.lines[0]?.account || undefined,
    80→    location: undefined,
    81→    memo: doc.memo,
    82→    dueDate: toDisplayDate(doc.dueDate),
    83→    amount: doc.total,
    84→    attachmentsCount: doc.attachments.length,
    85→    status: toTabStatus(doc.status),
    86→  };
    87→}
