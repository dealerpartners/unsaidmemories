   150→  onCreateCustomer,
   151→  onOpenCreateFlyout,
   152→  onToast,
   153→}: EstimatesScreenProps) {

   160→  const [menuOpen, setMenuOpen] = useState(false);

   163→

   172→

   174→  const [convertModalOpen, setConvertModalOpen] = useState(false);

   180→      setMenuOpen(false);
   181→      setNewMenuOpen(false);
   182→    };

   184→    return () => document.removeEventListener('click', handleGlobalClick);

   189→    const q = searchFilter.trim().toLowerCase();

   191→      const tabMatch =
   192→        activeTab === 'All' ||
   193→        (activeTab === 'Converted' ? est.status === 'Converted' : est.status === activeTab);
   194→      if (!tabMatch) return false;
   195→      if (!q) return true;
   196→      return (
   197→        est.no.toLowerCase().includes(q) ||
   198→        est.customer.toLowerCase().includes(q) ||
   199→        est.email.toLowerCase().includes(q) ||
   200→        est.status.toLowerCase().includes(q)
   201→      );
   202→    });
   203→  }, [estimates, activeTab, searchFilter]);
   204→

   206→  const handleCheckAll = (e: React.ChangeEvent<HTMLInputElement>) => {
   207→    if (e.target.checked) {
   208→      setSelectedIds(filteredEstimates.map((e) => e.id));
   209→    } else {
   210→      setSelectedIds([]);
   211→    }
   212→  };
   213→
   214→  const handleToggleCheck = (id: string, e: React.MouseEvent) => {
   215→    e.stopPropagation();
   216→    setSelectedIds((prev) =>
   217→      prev.includes(id) ? prev.filter((i) => i !== id) : [...prev, id]
   218→    );
   219→  };

   222→    const count = selectedIds.length;
   223→    if (action === 'delete') {
   224→      setEstimates((prev) => prev.filter((e) => !selectedIds.includes(e.id)));
   225→      setSelectedIds([]);
   226→      onToast?.(`Deleted ${count} estimate(s)`);
   227→      return;
   228→    }

   230→      setEstimates((prev) =>

   232→          selectedIds.includes(e.id)
   233→            ? { ...e, status: 'Converted', convertedInvoiceNo: `10${Math.floor(Math.random() * 90) + 10}` }
   234→            : e
   235→        )
   236→      );
   237→      setSelectedIds([]);
   238→      onToast?.(`Converted ${count} estimate(s) to invoice`);
   239→      return;
   240→    }
   241→    setSelectedIds([]);
   242→    onToast?.(`${action.charAt(0).toUpperCase() + action.slice(1)} on ${count} estimate(s)`);
   243→  };

   252→        id: `est-${Date.now()}`,

   263→        lines: [
   264→          { id: 'l1', desc: 'Custom design — 3 concepts', qty: 3, rate: 800.0 },
   265→          { id: 'l2', desc: 'Print setup + samples', qty: 1, rate: 400.0 },
   266→        ],
   267→      });
   268→    }
   269→    setDrawerOpen(true);
   270→  };
   271→
   272→  const handleDrawerSave = () => {
   273→    if (!editingEst) return;
   274→    const lineTotal = editingEst.lines.reduce((acc, l) => acc + l.qty * l.rate, 0);
   275→    const updated = { ...editingEst, amount: lineTotal };
   276→
   277→    setEstimates((prev) => {
   278→      const exists = prev.some((e) => e.id === updated.id);
   279→      if (exists) {
   280→        return prev.map((e) => (e.id === updated.id ? updated : e));
   281→      }
   282→      return [updated, ...prev];
   283→    });
   284→    setDrawerOpen(false);
   285→    onToast?.(`Estimate ${updated.no} saved`);
   286→  };
   287→
   288→  const handleDrawerSend = () => {
   289→    if (!editingEst) return;
   290→    const lineTotal = editingEst.lines.reduce((acc, l) => acc + l.qty * l.rate, 0);
   291→    const updated: EstimateItem = { ...editingEst, amount: lineTotal, status: 'Sent' };
   292→
   293→    setEstimates((prev) => {
   294→      const exists = prev.some((e) => e.id === updated.id);
   295→      if (exists) {
   296→        return prev.map((e) => (e.id === updated.id ? updated : e));
   297→      }
   298→      return [updated, ...prev];
   299→    });
   300→    setDrawerOpen(false);
   301→    onToast?.(`Estimate ${updated.no} sent to customer`);
   302→  };
   303→
   304→  const handlePromptConvert = (est: EstimateItem) => {
   305→    setConvertTargetEst(est);
   306→    setConvertModalOpen(true);
   307→  };

   309→  const handleConfirmConvert = () => {
   310→    if (!convertTargetEst) return;
   311→    const newInvoiceNo = `10${Math.floor(Math.random() * 90) + 10}`;
   312→    setEstimates((prev) =>
   313→      prev.map((e) =>
   314→        e.id === convertTargetEst.id
   315→          ? { ...e, status: 'Converted', convertedInvoiceNo: newInvoiceNo }
   316→          : e
   317→      )
   318→    );
   319→    setConvertModalOpen(false);
   320→    setDrawerOpen(false);
   321→    onToast?.(`Estimate ${convertTargetEst.no} converted to invoice ${newInvoiceNo}`);
   322→  };
   323→
   324→  // Row More Menu
   325→  const handleOpenRowMenu = (e: React.MouseEvent, est: EstimateItem) => {
   326→    e.stopPropagation();
   327→    const rect = e.currentTarget.getBoundingClientRect();
   328→    setMenuPos({
   329→      top: rect.bottom + 4,
   330→      left: Math.min(rect.right - 230, window.innerWidth - 250),
   331→    });
   332→    setActiveMenuEst(est);
   333→    setMenuOpen(true);
   334→  };
   335→
   336→  const handleRowMenuAction = (action: string) => {
   337→    if (!activeMenuEst) return;
   338→    const est = activeMenuEst;
   339→    setMenuOpen(false);
   340→
   341→    switch (action) {
   342→      case 'view':
   343→      case 'edit':
   344→        handleOpenDrawer(est);
   345→        break;
   346→      case 'duplicate': {
   347→        const copy: EstimateItem = {
   348→          ...est,
   349→          id: `est-${Date.now()}`,
   350→          no: `${est.no}-copy`,
   351→          status: 'Draft',
   352→        };
   353→        setEstimates((prev) => [copy, ...prev]);
   354→        onToast?.(`Duplicated estimate ${est.no}`);
   355→        break;
   356→      }
   357→      case 'send':
   358→        setEstimates((prev) =>
   359→          prev.map((e) => (e.id === est.id ? { ...e, status: 'Sent' } : e))
   360→        );
   361→        onToast?.(`Sent estimate ${est.no}`);
   362→        break;
   363→      case 'remind':
   364→        onToast?.(`Reminder sent for estimate ${est.no}`);
   365→        break;
   366→      case 'print':
   367→        onToast?.(`Printing estimate ${est.no} PDF`);
   368→        break;
   369→      case 'convert':
   370→        handlePromptConvert(est);
   371→        break;
   372→      case 'mark-accepted':
   373→        setEstimates((prev) =>
   374→          prev.map((e) => (e.id === est.id ? { ...e, status: 'Accepted' } : e))
   375→        );
   376→        onToast?.(`Estimate ${est.no} marked accepted`);
   377→        break;
   378→      case 'mark-declined':
   379→        setEstimates((prev) =>
   380→          prev.map((e) => (e.id === est.id ? { ...e, status: 'Declined' } : e))
   381→        );
   382→        onToast?.(`Estimate ${est.no} marked declined`);
   383→        break;
   384→      case 'delete':
   385→        setEstimates((prev) => prev.filter((e) => e.id !== est.id));
   386→        onToast?.(`Deleted estimate ${est.no}`);
   387→        break;

   417→        return <span className={`${styles.statusPill} ${styles.statusDeclined}`}>Declined</span>;

   419→        return <span className={`${styles.statusPill} ${styles.statusExpired}`}>Expired</span>;
   420→      default:
   421→        return <span className={`${styles.statusPill} ${styles.statusDraft}`}>{est.status}</span>;
   422→    }
   423→  };

   425→  return (
   426→    <div className={styles.container}>
   427→      {/* Context Header */}
   428→      <div className={styles.contextHeader}>
   429→        <div className={styles.breadcrumbGroup}>
   430→          <span className={`material-symbols-outlined ${styles.domainIcon}`}>
   431→            account_balance
   432→          </span>
   433→          <span className={styles.breadcrumbLabel}>Finance</span>
   434→          <span className={styles.breadcrumbSep}>/</span>
   435→          <span className={styles.breadcrumbLabel}>Sales &amp; Get Paid</span>
   436→          <span className={styles.breadcrumbSep}>/</span>
   437→          <span className={styles.breadcrumbActive}>Estimates</span>

   448→              Quotes sent to customers · convert to invoice when accepted
   449→            </p>
   450→          </div>
   451→
   452→          <div className={styles.headerActions}>
   453→            <div className={styles.splitBtn}>
   454→              <button
   455→                type="button"
   456→                className={styles.splitBtnMain}
   457→                onClick={() => handleOpenDrawer(null)}
   458→              >
   459→                <span className="material-symbols-outlined" style={{ fontSize: 16 }}>
   460→                  add
   461→                </span>
   462→                New estimate
   463→              </button>
   464→              <button
   465→                ref={newSplitBtnRef}
   466→                type="button"
   467→                className={styles.splitBtnArrow}
   468→                onClick={(e) => {
   469→                  e.stopPropagation();
   470→                  const rect = e.currentTarget.getBoundingClientRect();
   471→                  setNewMenuPos({
   472→                    top: rect.bottom + 4,
   473→                    left: Math.min(rect.right - 220, window.innerWidth - 240),
   474→                  });
   475→                  setNewMenuOpen((prev) => !prev);
   476→                }}
   477→              >
   478→                <span className="material-symbols-outlined" style={{ fontSize: 16 }}>
   479→                  expand_more
   480→                </span>
   481→              </button>
   482→            </div>
   483→
   484→            <button

   486→              className={styles.secondaryBtn}
   487→              onClick={() => {
   488→                const cols = [
   489→                  { key: 'no', label: 'Estimate No.' },
   490→                  { key: 'customer', label: 'Customer' },
   491→                  { key: 'email', label: 'Email' },
   492→                  { key: 'sentDate', label: 'Sent date' },
   493→                  { key: 'expiryDate', label: 'Expiry date' },

   500→            >

   503→              </span>
   504→              Export
   505→            </button>
   506→
   507→            <button
   508→              type="button"
   509→              className={styles.secondaryBtn}
   510→              onClick={() => onToast?.('Customize panel opened')}
   511→            >
   512→              <span className="material-symbols-outlined" style={{ fontSize: 16 }}>
   513→                tune
   514→              </span>
   515→              Customize
   516→            </button>
   517→          </div>
   518→        </div>

   520→        {/* KPI Strip */}
