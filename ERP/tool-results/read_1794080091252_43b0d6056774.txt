   195→/* ─── The Document Engine Component ─── */
   196→interface DocumentEngineProps {
   197→  type?: DocumentType;
   198→  documentId?: string | null; // null = new document
   199→  onToast: (msg: string) => void;
   200→  onClose?: () => void;
   201→  partyId?: string | null; // prefill from envelope
   202→  /** Open a different document (used by duplicate/convert to chain) */
   203→  onOpenDocument?: (id: string) => void;
   204→  /** Legacy controlled API (GlobalShell): open + activeTab + onSelectTab */
   205→  open?: boolean;
   206→  activeTab?: string;
   207→  onSelectTab?: (tab: string) => void;
   208→}
   209→
   210→export type DocTabType = 'edit' | 'email' | 'payor' | 'pdf';
   211→
   212→export default function DocumentEngine(props: DocumentEngineProps) {
   213→  // Legacy controlled mode (GlobalShell): open=false → render nothing.
   214→  if (props.open === false) return null;
   215→  return <DocumentEngineInner key={`${props.type || 'Invoice'}:${props.documentId || 'new'}`} {...props} />;
   216→}
   217→
   218→function DocumentEngineInner({ type = 'Invoice', documentId, onToast, onClose, partyId: prefillPartyId, activeTab, onSelectTab, onOpenDocument }: DocumentEngineProps) {
   219→  const config = CONFIGS[type] || CONFIGS.Invoice;
   220→  const store = usePlatformStore();
   221→
   222→  const existing = documentId ? store.getDocument(documentId) : null;
   223→  const isNew = !existing;
   224→
   225→  // §1.3 Context Envelope — capture once at mount. Creating from a record
   226→  // prefills from that record; creating from the global menu inherits the
   227→  // active tab's envelope.
   228→  const initialEnv = useRef(usePlatformStore.getState().envelope).current;
   229→
   230→  // Form state
   231→  const [number, setNumber] = useState(existing?.number || `${config.numberPrefix}${String(store.documents.filter(d => d.type === type).length + 1001)}`);
   232→  const [partyId, setPartyId] = useState(existing?.partyId || prefillPartyId || initialEnv.partyId || '');
   233→  const [status, setStatus] = useState(existing?.status || 'Draft');
   234→  const [date, setDate] = useState(existing?.date || new Date().toISOString().slice(0, 10));
   235→  const [dueDate, setDueDate] = useState(existing?.dueDate || '');
   236→  const [lines, setLines] = useState<DocumentLine[]>(existing?.lines || [{ id: 'L1', itemId: null, description: '', qty: 1, rate: 0, amount: 0, taxRate: 0, account: '', tags: [] }]);
   237→  const [shipping, setShipping] = useState(existing?.shipping || 0);
   238→  const [memo, setMemo] = useState(existing?.memo || '');
   239→  const [notes, setNotes] = useState(existing?.notes || '');
   240→  const [viewState, setViewState] = useState<'edit' | 'email' | 'payor' | 'pdf'>((activeTab as 'edit' | 'email' | 'payor' | 'pdf') || 'edit');
   241→  const [showPaymentModal, setShowPaymentModal] = useState(false);
   242→  const [payAmount, setPayAmount] = useState('');
   243→  const [payMethod, setPayMethod] = useState('Cash');
   244→  const [payAccount, setPayAccount] = useState<string | null>(null);
   245→
   246→  // §1.3: "On Order #1042 → Create → Invoice: party, lines, channel, and
   247→  // order_id are prefilled and linked."
   248→  useEffect(() => {
   249→    if (existing || prefillPartyId) return;
   250→    if (initialEnv.orderId) {
   251→      const order = usePlatformStore.getState().getDocument(initialEnv.orderId);
   252→      if (order) {
   253→        setPartyId(order.partyId);
   254→        if (order.lines.length > 0) setLines(order.lines.map((l, i) => ({ ...l, id: `L${i + 1}` })));
   255→      }
   256→    }
   257→  }, []);
   258→
   259→  // Legacy tab sync
   260→  const setView = (v: 'edit' | 'email' | 'payor' | 'pdf') => { setViewState(v); onSelectTab?.(v); };
   261→
   262→  const party = partyId ? store.getParty(partyId) : null;
   263→  
   264→  const subtotal = useMemo(() => lines.reduce((s, l) => s + l.amount, 0), [lines]);
   265→  const tax = useMemo(() => lines.reduce((s, l) => s + (l.amount * l.taxRate / 100), 0), [lines]);
   266→  const total = subtotal + tax + shipping;
   267→  const balance = existing ? existing.balance : total;
   268→
   269→  function updateLine(idx: number, updates: Partial<DocumentLine>) {
   270→    setLines(prev => prev.map((l, i) => {
   271→      if (i !== idx) return l;
   272→      const updated = { ...l, ...updates };
   273→      updated.amount = updated.qty * updated.rate;
   274→      return updated;
   275→    }));
   276→  }
   277→
   278→  function addLine() {
   279→    setLines(prev => [...prev, { id: `L${prev.length + 1}`, itemId: null, description: '', qty: 1, rate: 0, amount: 0, taxRate: 0, account: '', tags: [] }]);
   280→  }
   281→
   282→  function removeLine(idx: number) {
   283→    setLines(prev => prev.filter((_, i) => i !== idx));
   284→  }
   285→
   286→  function save(andNew = false) {
   287→    if (!partyId && config.partyRole) { onToast(`${config.partyLabel} is required`); return; }
   288→    const docData = {
   289→      type, number, partyId, status, date, dueDate, lines, subtotal, tax: Math.round(tax * 100) / 100, shipping, total, amountPaid: existing?.amountPaid || 0, balance: total - (existing?.amountPaid || 0), memo, notes, tags: existing?.tags || [], attachments: existing?.attachments || [],
   290→      // §1.3 Context Envelope: creating from a record links back to it
   291→      orderId: existing?.orderId || (isNew ? initialEnv.orderId : null),
   292→      poId: existing?.poId || null,
   293→      channelId: existing?.channelId || (isNew ? initialEnv.channelId : null),
   294→      createdBy: 'Current user',
   295→    };
   296→    if (existing) {
   297→      store.updateDocument(existing.id, docData);
   298→      store.recalcDocument(existing.id);
   299→      onToast(`${config.label} saved`);
   300→      if (!andNew) onClose?.();
   301→    } else {
   302→      const id = store.addDocument(docData as Omit<Document, 'id' | 'createdAt' | 'updatedAt' | 'activity'>);
   303→      onToast(`${config.label} ${number} created`);
   304→      if (andNew) {
   305→        // Save & new: reset the form for the next document
   306→        setNumber(`${config.numberPrefix}${String(store.documents.filter(d => d.type === type).length + 1002)}`);
   307→        setPartyId(initialEnv.partyId || '');
   308→        setLines([{ id: 'L1', itemId: null, description: '', qty: 1, rate: 0, amount: 0, taxRate: 0, account: '', tags: [] }]);
   309→        setShipping(0); setMemo(''); setNotes('');
   310→      } else {
   311→        onClose?.();
   312→      }
   313→    }
   314→  }
   315→
   316→  /* §1.7 Duplicate — "Edit · Duplicate" on every document */
   317→  function duplicate() {
   318→    if (!existing) return;
   319→    const newId = store.duplicateDocument(existing.id);
   320→    if (newId) {
   321→      onToast(`${config.label} duplicated`);
   322→      if (onOpenDocument) onOpenDocument(newId);
   323→      else onClose?.();
   324→    }
   325→  }
   326→
   327→  /* Estimate → Invoice (per doc: "button Convert to invoice") */
   328→  function convertToInvoice() {
   329→    if (!existing) return;
   330→    const newId = store.convertDocument(existing.id, 'Invoice');
   331→    if (newId) {
   332→      onToast(`${existing.number} converted to invoice`);
   333→      if (onOpenDocument) onOpenDocument(newId);
   334→      else onClose?.();
   335→    }
   336→  }
   337→
   338→  function send() {
   339→    if (!existing && isNew) { save(); return; }
   340→    if (existing) {
   341→      store.changeDocumentStatus(existing.id, 'Sent' as any);
   342→      onToast(`${config.label} ${number} sent to ${party?.email || 'customer'}`);
   343→    }
   344→  }
   345→
   346→  function openPaymentModal() {
   347→    setPayAmount(balance.toFixed(2));
   348→    const bank = store.getAccountsByType('Bank')[0];
   349→    setPayAccount(bank ? bank.id : null);
   350→    setShowPaymentModal(true);
   351→  }
   352→
   353→  function recordPayment() {
   354→    const amount = parseFloat(payAmount || '0');
   355→    if (!existing) { onToast('Save the document first'); return; }
   356→    if (!amount || amount <= 0) { onToast('Enter a payment amount'); return; }
   357→    const accountName = payAccount ? (store.getAccount(payAccount)?.name || 'Checking') : 'Checking';
   358→    store.recordPayment(existing.id, amount, payMethod, accountName);
   359→    onToast(`Payment of $${amount.toFixed(2)} recorded via ${payMethod}`);
   360→    setShowPaymentModal(false);
   361→  }
   362→
   363→  return (
   364→    <div className="flex flex-col h-full">
   365→      {/* Header */}
   366→      <header className="h-12 bg-white border-b border-surface-dim flex items-center justify-between px-4 shrink-0">
   367→        <div className="flex items-center gap-1">
   368→          {(['edit', 'email', 'payor', 'pdf'] as const).map(v => (
   369→            <button key={v} className={`px-3 py-2 text-[13px] tab-underline ${viewState === v ? 'active' : 'text-on-surface-variant'}`} onClick={() => setView(v)}>
   370→              {v === 'edit' ? 'Edit' : v === 'email' ? 'Email view' : v === 'payor' ? 'Payor view' : 'PDF view'}
   371→            </button>
   372→          ))}
   373→        </div>
   374→        <div className="flex items-center gap-3 text-[13px] text-on-surface-variant">
   375→          {onClose && <button className="icon-btn" onClick={onClose}><span className="material-symbols-outlined">close</span></button>}
   376→        </div>
   377→      </header>
   378→
   379→      {/* Main */}
   380→      <div className="flex-1 flex overflow-hidden bg-[#faf9fc]">
   381→        {viewState === 'edit' && (
   382→          <div className="flex-1 overflow-y-auto p-6 flex justify-center">
   383→            <div className="w-full max-w-[900px] space-y-4">
   384→              {/* Document paper */}
   385→              <div className="bg-white border border-surface-dim shadow-sm rounded-sm min-h-[600px] p-8">
   386→                {/* Header row */}
   387→                <div className="flex justify-between items-start mb-6">
   388→                  <div>
   389→                    <h1 className="text-[24px] font-bold text-primary mb-1">{config.label.toUpperCase()}</h1>
   390→                    <div className="text-[13px] font-semibold">{number}</div>
   391→                  </div>
   392→                  <div className="text-right">
   393→                    <div className="text-[12px] text-on-surface-variant uppercase font-bold mb-1">{config.label} total</div>
   394→                    <div className="text-[24px] font-bold">${total.toFixed(2)}</div>
   395→                  </div>
   396→                </div>
   397→
   398→                {/* Party + dates */}
   399→                <div className="grid grid-cols-2 gap-8 mb-6">
   400→                  <div>
   401→                    {config.partyRole ? (
   402→                      <PartyPicker
   403→                        value={partyId || null}
   404→                        onChange={id => setPartyId(id || '')}
   405→                        role={config.partyRole as PartyRole}
   406→                        label={config.partyLabel}
   407→                        required
   408→                      />
   409→                    ) : (
   410→                      <div>
   411→                        <label className="block text-[11px] text-on-surface-variant uppercase font-bold mb-1">{config.partyLabel}</label>
   412→                        <div className="text-[13px] text-on-surface-variant italic">No party required</div>
   413→                      </div>
   414→                    )}
   415→                    {party && (
   416→                      <div className="mt-2 text-[12px] text-on-surface-variant">
   417→                        {party.address}, {party.city}, {party.state} {party.zip}
   418→                      </div>
   419→                    )}
   420→                  </div>
   421→                  <div className="space-y-3">
   422→                    <div>
   423→                      <label className="block text-[11px] text-on-surface-variant uppercase font-bold mb-1">{config.dateLabel}</label>
   424→                      <input type="date" className="field-input" value={date} onChange={e => setDate(e.target.value)} />
   425→                    </div>

   607→/* ─── LineItemSearch — §1.4 Item picker on document lines ───
   608→   "Search by name/SKU/barcode, shows price, cost, stock pill".
   609→   Typing shows matching items; picking one fills description + rate
   610→   (cost for Bill/PO/Vendor credit, price otherwise). Free text stays
   611→   valid for Blank/Service lines. */
   612→function LineItemSearch({ line, costBased, onPick, onText }: {
   613→  line: DocumentLine;
   614→  costBased: boolean;
   615→  onPick: (item: Item) => void;
   616→  onText: (text: string) => void;
   617→}) {
   618→  const items = usePlatformStore(s => s.items);
   619→  const [open, setOpen] = useState(false);
   620→  const ref = useRef<HTMLDivElement>(null);
   621→
   622→  useEffect(() => {
   623→    if (!open) return;
   624→    const h = (e: MouseEvent) => { if (ref.current && !ref.current.contains(e.target as Node)) setOpen(false); };
   625→    document.addEventListener('mousedown', h);
   626→    return () => document.removeEventListener('mousedown', h);
   627→  }, [open]);
   628→
   629→  const q = line.description.toLowerCase().trim();
   630→  const matches = q
   631→    ? items.filter(i => (i.name + ' ' + i.sku + ' ' + i.category).toLowerCase().includes(q)).slice(0, 6)
   632→    : [];
   633→
   634→  return (
   635→    <div ref={ref} className="relative">
   636→      <div className="flex items-center gap-1">
   637→        {line.itemId && <span className="material-symbols-outlined text-[14px] text-primary" title="Linked catalog item">inventory_2</span>}
   638→        <input
   639→          className="w-full border-0 bg-transparent text-[13px] focus:outline-none focus:border-b focus:border-primary"
   640→          placeholder="Type to search items or enter description"
   641→          value={line.description}
   642→          onChange={e => { onText(e.target.value); setOpen(true); }}
   643→          onFocus={() => setOpen(true)}
   644→        />
   645→      </div>
   646→      {open && matches.length > 0 && (
   647→        <div className="row-menu open" style={{ position: 'absolute', top: '100%', left: 0, minWidth: 280, zIndex: 9999 }}>
   648→          {matches.map(i => (
   649→            <button key={i.id} type="button" className="w-full text-left px-3 py-2 hover:bg-surface-container-low flex items-center gap-2" onClick={() => { onPick(i); setOpen(false); }}>
   650→              <div className="flex-1 min-w-0">
   651→                <div className="text-[13px] font-medium truncate">{i.name}</div>
   652→                <div className="text-[10px] text-on-surface-variant">{i.sku} · {costBased ? `cost $${i.cost.toFixed(2)}` : `$${i.price.toFixed(2)}`}</div>
   653→              </div>
   654→              {i.trackInventory && <span className={`pill ${i.stockOnHand <= i.reorderPoint ? 'bad' : 'ok'}`} style={{ fontSize: 8, padding: '1px 4px' }}>{i.stockOnHand}</span>}
   655→              <span className="pill role" style={{ fontSize: 8, padding: '1px 4px' }}>{i.type}</span>
   656→            </button>
   657→          ))}
   658→        </div>
   659→      )}
   660→    </div>
   661→  );
   662→}
