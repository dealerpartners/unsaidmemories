   297→   297→
   298→   298→interface PlatformState {
   299→   299→  // Entities
   300→   300→  parties: Party[];
   301→   301→  items: Item[];
   302→   302→  documents: Document[];
   303→   303→  moneyMovements: MoneyMovement[];
   304→   304→  accounts: Account[];
   305→   305→
   306→   306→  // Context Envelope (§1.3)
   307→   307→  envelope: {
   308→   308→    originModule: string | null;
   309→   309→    originRecordId: string | null;
   310→   310→    partyId: string | null;
   311→   311→    orderId: string | null;
   312→   312→    itemId: string | null;
   313→   313→    designId: string | null;
   314→   314→    accountId: string | null;
   315→   315→    channelId: string | null;
   316→   316→    returnTo: string | null;
   317→   317→  };
   318→   318→
   319→   319→  // Party CRUD
   320→   320→  addParty: (party: Omit<Party, 'id' | 'added' | 'lastActivity'>) => string;
   321→   321→  updateParty: (id: string, updates: Partial<Party>) => void;
   322→   322→  deleteParty: (id: string) => void;
   323→   323→  getParty: (id: string) => Party | undefined;
   324→   324→  getPartiesByRole: (role: PartyRole) => Party[];
   325→   325→
   326→   326→  // Item CRUD
   327→   327→  addItem: (item: Omit<Item, 'id' | 'updated'>) => string;
   328→   328→  updateItem: (id: string, updates: Partial<Item>) => void;
   329→   329→  deleteItem: (id: string) => void;
   330→   330→  getItem: (id: string) => Item | undefined;
   331→   331→
   332→   332→  // Document CRUD
   333→   333→  addDocument: (doc: Omit<Document, 'id' | 'createdAt' | 'updatedAt' | 'activity'>) => string;
   334→   334→  updateDocument: (id: string, updates: Partial<Document>) => void;
   335→   335→  deleteDocument: (id: string) => void;
   336→   336→  getDocument: (id: string) => Document | undefined;
   337→   337→  getDocumentsByType: (type: DocumentType) => Document[];
   338→   338→  getDocumentsByParty: (partyId: string) => Document[];
   339→   339→  recordPayment: (documentId: string, amount: number, method: string, fromAccount: string) => void;
   340→   340→  changeDocumentStatus: (id: string, status: DocumentStatus) => void;
   341→   341→  recalcDocument: (id: string) => void;
   342→   342→
   343→   343→  // Money movement CRUD
   344→   344→  addMoney: (money: Omit<MoneyMovement, 'id'>) => string;
   345→   345→
   346→   346→  // Account CRUD
   347→   347→  getAccount: (id: string) => Account | undefined;
   348→   348→  getAccountsByType: (type: AccountType) => Account[];
   349→   349→
   350→   350→  // Envelope
   351→   351→  setEnvelope: (env: Partial<PlatformState['envelope']>) => void;
   352→   352→  clearEnvelope: () => void;
   353→   353→}

   355→   355→let idCounter = Date.now();
   356→   356→const nextId = (prefix: string) => `${prefix}-${++idCounter}`;
   357→   357→
   358→   358→export const usePlatformStore = create<PlatformState>()(
   359→   359→  persist(
   360→   360→    (set, get) => ({
   361→   361→      parties: SEED_PARTIES,
   362→   362→      items: SEED_ITEMS,
   363→   363→      documents: SEED_DOCUMENTS,
   364→   364→      moneyMovements: SEED_MONEY,
   365→   365→      accounts: SEED_ACCOUNTS,
   366→   366→
   367→   367→      envelope: {
   368→   368→        originModule: null, originRecordId: null, partyId: null,
   369→   369→        orderId: null, itemId: null, designId: null,
   370→   370→        accountId: null, channelId: null, returnTo: null,
   371→   371→      },
   372→   372→
   373→   373→      // ── Party CRUD ──
   374→   374→      addParty: (party) => {
   375→   375→        const id = nextId('P');
   376→   376→        const newParty: Party = {
   377→   377→          ...party, id, added: new Date().toISOString().slice(0, 10),
   378→   378→          lastActivity: new Date().toISOString().slice(0, 10),
   379→   379→        };
   380→   380→        set(state => ({ parties: [newParty, ...state.parties] }));
   381→   381→        return id;
   382→   382→      },
   383→   383→      updateParty: (id, updates) => {
   384→   384→        set(state => ({
   385→   385→          parties: state.parties.map(p => p.id === id ? { ...p, ...updates, lastActivity: new Date().toISOString().slice(0, 10) } : p),
   386→   386→        }));
   387→   387→      },
   388→   388→      deleteParty: (id) => {
   389→   389→        set(state => ({ parties: state.parties.filter(p => p.id !== id) }));
   390→   390→      },
   391→   391→      getParty: (id) => get().parties.find(p => p.id === id),
   392→   392→      getPartiesByRole: (role) => get().parties.filter(p => p.roles.includes(role)),
   393→   393→
   394→   394→      // ── Item CRUD ──
   395→   395→      addItem: (item) => {
   396→   396→        const id = nextId('ITM');
   397→   397→        const newItem: Item = { ...item, id, updated: new Date().toISOString().slice(0, 10) };
   398→   398→        set(state => ({ items: [newItem, ...state.items] }));
   399→   399→        return id;
   400→   400→      },
   401→   401→      updateItem: (id, updates) => {
   402→   402→        set(state => ({
   403→   403→          items: state.items.map(i => i.id === id ? { ...i, ...updates, updated: new Date().toISOString().slice(0, 10) } : i),
   404→   404→        }));
   405→   405→      },
   406→   406→      deleteItem: (id) => {
   407→   407→        set(state => ({ items: state.items.filter(i => i.id !== id) }));
   408→   408→      },
   409→   409→      getItem: (id) => get().items.find(i => i.id === id),
   410→   410→
   411→   411→      // ── Document CRUD ──
   412→   412→      addDocument: (doc) => {
   413→   413→        const id = nextId('DOC');
   414→   414→        const now = new Date().toISOString();
   415→   415→        const newDoc: Document = {
   416→   416→          ...doc, id, createdAt: now, updatedAt: now,
   417→   417→          activity: [{ date: now.slice(0, 10), type: 'created', text: `${doc.type} created`, actor: 'Current user' }],
   418→   418→        };
   419→   419→        set(state => ({ documents: [newDoc, ...state.documents] }));
   420→   420→        return id;
   421→   421→      },
   422→   422→      updateDocument: (id, updates) => {
   423→   423→        const now = new Date().toISOString();
   424→   424→        set(state => ({
   425→   425→          documents: state.documents.map(d => d.id === id ? { ...d, ...updates, updatedAt: now } : d),
   426→   426→        }));
   427→   427→      },
   428→   428→      deleteDocument: (id) => {
   429→   429→        set(state => ({ documents: state.documents.filter(d => d.id !== id) }));
   430→   430→      },
   431→   431→      getDocument: (id) => get().documents.find(d => d.id === id),
   432→   432→      getDocumentsByType: (type) => get().documents.filter(d => d.type === type),
   433→   433→      getDocumentsByParty: (partyId) => get().documents.filter(d => d.partyId === partyId),
   434→   434→
   435→   435→      recordPayment: (documentId, amount, method, fromAccount) => {
   436→   436→        const doc = get().documents.find(d => d.id === documentId);
   437→   437→        if (!doc) return;
   438→   438→        const newAmountPaid = doc.amountPaid + amount;
   439→   439→        const newBalance = doc.total - newAmountPaid;
   440→   440→        const newStatus = newBalance <= 0 ? 'Paid' : 'Unpaid';
   441→   441→        const now = new Date().toISOString();
   442→   442→        const payId = nextId('PAY');
   443→   443→        const payment: MoneyMovement = {
   444→   444→          id: payId, type: 'Payment', amount, date: now.slice(0, 10),
   445→   445→          method, fromAccount: fromAccount || 'Customer', toAccount: '1000 Checking',
   446→   446→          partyId: doc.partyId, documentId, reference: '', status: 'Cleared',
   447→   447→          memo: `Payment for ${doc.number}`, createdBy: 'Current user',
   448→   448→        };
   449→   449→        set(state => ({
   450→   450→          documents: state.documents.map(d => d.id === documentId ? {
   451→   451→            ...d, amountPaid: newAmountPaid, balance: newBalance, status: newStatus,
   452→   452→            updatedAt: now,
   453→   453→            activity: [...d.activity, { date: now.slice(0, 10), type: 'payment', text: `Payment received $${amount.toFixed(2)} via ${method}`, actor: 'Current user' }],
   454→   454→          } : d),
   455→   455→          moneyMovements: [payment, ...state.moneyMovements],
   456→   456→          parties: state.parties.map(p => p.id === doc.partyId ? { ...p, balance: p.balance - amount } : p),
   457→   457→        }));
   458→   458→      },
   459→   459→
   460→   460→      changeDocumentStatus: (id, status) => {
   461→   461→        const now = new Date().toISOString();
   462→   462→        set(state => ({
   463→   463→          documents: state.documents.map(d => d.id === id ? {
   464→   464→            ...d, status, updatedAt: now,
   465→   465→            activity: [...d.activity, { date: now.slice(0, 10), type: 'status', text: `Status changed to ${status}`, actor: 'Current user' }],
   466→   466→          } : d),
   467→   467→        }));
   468→   468→      },
   469→   469→
   470→   470→      recalcDocument: (id) => {
   471→   471→        set(state => ({
   472→   472→          documents: state.documents.map(d => {
   473→   473→            if (d.id !== id) return d;
   474→   474→            const subtotal = d.lines.reduce((s, l) => s + l.amount, 0);
   475→   475→            const tax = d.lines.reduce((s, l) => s + (l.amount * l.taxRate / 100), 0);
   476→   476→            const total = subtotal + tax + d.shipping;
   477→   477→            const balance = total - d.amountPaid;
   478→   478→            return { ...d, subtotal, tax: Math.round(tax * 100) / 100, total, balance };
   479→   479→          }),
   480→   480→        }));
   481→   481→      },
   482→   482→
   483→   483→      // ── Money movement ──
   484→   484→      addMoney: (money) => {
   485→   485→        const id = nextId('PAY');
   486→   486→        set(state => ({ moneyMovements: [{ ...money, id }, ...state.moneyMovements] }));
   487→   487→        return id;
   488→   488→      },
   489→   489→
   490→   490→      // ── Account ──
   491→   491→      getAccount: (id) => get().accounts.find(a => a.id === id),
   492→   492→      getAccountsByType: (type) => get().accounts.filter(a => a.type === type),
   493→   493→
   494→   494→      // ── Envelope ──
   495→   495→      setEnvelope: (env) => {
   496→   496→        set(state => ({ envelope: { ...state.envelope, ...env } }));
   497→   497→      },
   498→   498→      clearEnvelope: () => {
   499→   499→        set({
   500→   500→          envelope: {
   501→   501→            originModule: null, originRecordId: null, partyId: null,
   502→   502→            orderId: null, itemId: null, designId: null,
   503→   503→            accountId: null, channelId: null, returnTo: null,
   504→   504→          },
   505→   505→        });
   506→   506→      },
   507→   507→    }),
   508→   508→    {
   509→   509→      name: 'platform-store',
   510→   510→      // Only persist the data, not the functions
   511→   511→      partialize: (state) => ({
   512→   512→        parties: state.parties,
   513→   513→        items: state.items,
   514→   514→        documents: state.documents,
   515→   515→        moneyMovements: state.moneyMovements,
   516→   516→        accounts: state.accounts,
   517→   517→      }),
   518→   518→    }
   519→   519→  )
   520→   520→);
