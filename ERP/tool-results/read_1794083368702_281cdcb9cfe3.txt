     1→'use client';
     2→/* StatementsScreen — 1:1 port of upload/Contacts_extracted/Contacts/statements.html
     3→   Now derived from the shared store: a statement is the party's invoice
     4→   activity for a period (§Sales: "Statement | Workspace (batch)"). */
     5→import { useMemo, useState } from 'react';
     6→import { exportToCsv } from '@/lib/csvExport';
     7→import { usePlatformStore } from '@/lib/platformStore';
     8→
     9→interface StatementsScreenProps { onToast: (m: string) => void; onNavigate?: (d: string, s: string) => void; }
    10→
    11→interface StmtLine { date: string; desc: string; amt: string; paid: string; bal: string; }
    12→interface Stmt {
    13→  id: string; customer: string; email: string; period: string; date: string; due: string;
    14→  invoiced: string; paid: string; dueAmt: string; status: string; balance: string; lines: StmtLine[];
    15→  avatarColor: string; avatarLetter: string;
    16→  partyId: string;
    17→}
    18→
    19→const AVATAR_COLORS = ['bg-qb-blue', 'bg-qb-teal', 'bg-qb-green', 'bg-qb-maroon', 'bg-qb-navy'];
    20→const money = (n: number) => `$${n.toFixed(2)}`;
    21→const monthOf = (iso: string) => { const d = new Date(iso + 'T00:00:00'); return isNaN(d.getTime()) ? '—' : d.toLocaleDateString('en-US', { month: 'short', year: 'numeric' }); };
    22→
    23→function statusClass(s: string) { return s === 'Sent' ? 'sent' : s === 'Overdue' ? 'overdue' : s === 'Paid' ? 'paid' : s === 'Draft' ? 'draft' : s === 'Partial' ? 'partial' : 'sent'; }
    24→
    25→export default function StatementsScreen({ onToast }: StatementsScreenProps) {
    26→  const store = usePlatformStore();
    27→
    28→  /* A statement = one customer's invoice activity for their most recent
    29→     period, derived live from the shared store. */
    30→  const items: Stmt[] = useMemo(() => {
    31→    const customers = store.parties.filter(p => p.roles.includes('Customer'));
    32→    return customers.map((p, ci) => {
    33→      const invoices = store.documents.filter(d => d.partyId === p.id && d.type === 'Invoice');
    34→      if (invoices.length === 0) return null;
    35→      const invoiced = invoices.reduce((s, d) => s + d.total, 0);
    36→      const paid = invoices.reduce((s, d) => s + d.amountPaid, 0);
    37→      const due = invoices.reduce((s, d) => s + d.balance, 0);
    38→      const hasOverdue = invoices.some(d => d.status === 'Overdue');
    39→      const hasDraft = invoices.some(d => d.status === 'Draft');
    40→      const status = hasOverdue ? 'Overdue' : hasDraft ? 'Draft' : due <= 0 ? 'Paid' : paid > 0 ? 'Partial' : 'Sent';
    41→      const latest = invoices.reduce((a, b) => (a.date > b.date ? a : b));
    42→      const period = monthOf(latest.date);
    43→      return {
    44→        id: `ST-${p.id.slice(-4)}`, partyId: p.id,
    45→        customer: p.name, email: p.email,
    46→        period, date: latest.date, due: latest.dueDate || latest.date,
    47→        invoiced: money(invoiced), paid: `−${money(paid)}`, dueAmt: money(due),
    48→        status, balance: money(due),
    49→        lines: invoices.map(d => ({ date: d.date.slice(5).replace('-', '/'), desc: `Invoice ${d.number}`, amt: money(d.total), paid: money(d.amountPaid), bal: money(d.balance) })),
    50→        avatarColor: AVATAR_COLORS[ci % AVATAR_COLORS.length], avatarLetter: (p.name[0] || '?').toUpperCase(),
    51→      };
    52→    }).filter((s): s is Stmt => s !== null);
    53→  }, [store.parties, store.documents]);
    54→
    55→  const [tab, setTab] = useState('All');
    56→  const [q, setQ] = useState('');
    57→  const [selected, setSelected] = useState<Set<string>>(new Set());
    58→  const [drawerId, setDrawerId] = useState<string | null>(null);
    59→  const [rowMenuForId, setRowMenuForId] = useState<string | null>(null);
    60→  const [newMenuOpen, setNewMenuOpen] = useState(false);
    61→  const [batchOpen, setBatchOpen] = useState(false);
    62→  const [hidden, setHidden] = useState<Set<string>>(new Set());
    63→
    64→  const filtered = useMemo(() => items.filter(s => {
    65→    if (hidden.has(s.id)) return false;
    66→    const tabOK = tab === 'All' || s.status === tab || (tab === 'Awaiting payment' && (s.status === 'Sent' || s.status === 'Partial'));
    67→    const qOK = !q || (s.customer + ' ' + s.period + ' ' + s.id).toLowerCase().includes(q.toLowerCase());
    68→    return tabOK && qOK;
    69→  }), [items, tab, q, hidden]);
    70→
    71→  const allChecked = filtered.length > 0 && filtered.every(s => selected.has(s.id));
    72→  const someChecked = filtered.some(s => selected.has(s.id));
    73→  const drawerStmt = drawerId ? items.find(s => s.id === drawerId) : null;
    74→
    75→  function toggleRow(id: string) { setSelected(prev => { const n = new Set(prev); if (n.has(id)) n.delete(id); else n.add(id); return n; }); }
    76→  function toggleAll() { if (allChecked) setSelected(new Set()); else setSelected(new Set(filtered.map(s => s.id))); }
    77→
    78→  function rowAction(act: string, s: Stmt) {
    79→    setRowMenuForId(null);
    80→    switch (act) {
    81→      case 'view': case 'edit': setDrawerId(s.id); break;
    82→      case 'duplicate': onToast(`Statement regenerated for ${s.customer} — reopens with latest invoice activity`); break;
    83→      case 'send': onToast(`Statement sent to ${s.customer} (${s.email})`); break;
    84→      case 'remind': onToast(`Reminder sent for ${s.id} to ${s.customer}`); break;
    85→      case 'print': window.print(); break;
    86→      case 'mark-paid': {
    87→        // Real money movement: record payment against each open invoice
    88→        const open = store.documents.filter(d => d.partyId === s.partyId && d.type === 'Invoice' && d.balance > 0);
    89→        open.forEach(d => store.recordPayment(d.id, d.balance, 'Check', 'Checking'));
    90→        onToast(`${s.customer} marked as paid — ${open.length} payment${open.length !== 1 ? 's' : ''} recorded`);
    91→        break;
    92→      }
    93→      case 'receive-payment': onToast(`Receive payment for ${s.customer} — use the Receive payment workspace`); break;
    94→      case 'delete': setHidden(prev => { const n = new Set(prev); n.add(s.id); return n; }); onToast(`Statement ${s.id} removed from the list`); break;
    95→    }
    96→  }
    97→
    98→  function bulkAction(act: string) {
    99→    if (act === 'close') { setSelected(new Set()); return; }
   100→    const n = selected.size;
   101→    if (act === 'delete') { setHidden(prev => new Set([...prev, ...selected])); }
   102→    if (act === 'send') { onToast(`Statements sent to ${n} customers`); }
   103→    else onToast(`${act.charAt(0).toUpperCase() + act.slice(1)} on ${n} statement${n !== 1 ? 's' : ''}`);
   104→    setSelected(new Set());
   105→  }
   106→
   107→  function handleExport() {
   108→    exportToCsv(filtered as unknown as Record<string, unknown>[], [
   109→      { key: 'id', label: 'Statement ID' }, { key: 'customer', label: 'Customer' }, { key: 'period', label: 'Period' },
   110→      { key: 'date', label: 'Sent' }, { key: 'due', label: 'Due' }, { key: 'balance', label: 'Balance' }, { key: 'status', label: 'Status' },
   111→    ], 'statements');
   112→    onToast(`Exported ${filtered.length} statements`);
   113→  }
   114→
   115→  const tabs = ['All', 'Draft', 'Sent', 'Awaiting payment', 'Paid', 'Overdue'];
   116→
   117→  return (
   118→    <main className="px-6 pb-12">
   119→      {/* BREADCRUMB */}
   120→      <div className="flex items-center gap-1 text-label-sm text-on-surface-variant mt-4 mb-1">
   121→        <span>Contacts</span><span className="text-outline">/</span><span className="text-label-md">Customer Hub</span><span className="text-outline">/</span><span className="text-label-md text-primary font-semibold">Statements</span>
   122→      </div>
   123→
   124→      {/* PAGE HEADER */}
   125→      <div className="flex items-center justify-between gap-4 py-4 flex-wrap">
   126→        <div><h1 className="text-headline-lg">Statements</h1><p className="text-body-sm text-on-surface-variant mt-0.5">Batch billing summary — send a customer their account activity for a period</p></div>
   127→        <div className="flex items-center gap-2">
   128→          <div className="flex split-btn">
   129→            <button className="h-8 px-3 bg-primary text-white text-body-md font-semibold hover:bg-primary-container flex items-center gap-1 split-btn-main" onClick={() => setDrawerId('__new__')}><span className="material-symbols-outlined text-[16px]">add</span>New statement</button>
   130→            <button className="h-8 px-1.5 bg-primary text-white hover:bg-primary-container split-btn-arrow" onClick={() => setNewMenuOpen(v => !v)}><span className="material-symbols-outlined text-[16px]">expand_more</span></button>
   131→          </div>
   132→          <button className="h-8 px-3 border border-surface-dim rounded text-body-md hover:bg-surface-container-low flex items-center gap-1" onClick={handleExport}><span className="material-symbols-outlined text-[16px]">download</span>Export</button>
   133→          <button className="h-8 px-3 border border-surface-dim rounded text-body-md hover:bg-surface-container-low flex items-center gap-1" onClick={() => onToast('Customize panel')}><span className="material-symbols-outlined text-[16px]">tune</span>Customize</button>
   134→        </div>
   135→      </div>
   136→
   137→      {/* KPI STRIP */}
   138→      <div className="grid grid-cols-2 xl:grid-cols-4 gap-3 mb-5">
   139→        <div className="bg-white rounded-xl border border-surface-dim p-3"><div className="text-[10px] text-on-surface-variant uppercase tracking-wider font-bold">Statements</div><div className="text-headline-sm mt-1 tabular-nums">{items.length}</div><div className="text-[11px] text-on-surface-variant mt-0.5">{money(items.reduce((s, x) => s + (parseFloat(x.invoiced.replace(/[$,]/g, '')) || 0), 0))} invoiced total</div></div>
   140→        <div className="bg-white rounded-xl border border-surface-dim p-3"><div className="text-[10px] text-on-surface-variant uppercase tracking-wider font-bold">Awaiting payment</div><div className="text-headline-sm mt-1 tabular-nums">{items.filter(s => s.status === 'Sent' || s.status === 'Partial').length}</div><div className="text-[11px] text-on-surface-variant mt-0.5">{money(items.reduce((s, x) => s + (parseFloat(x.dueAmt.replace(/[$,]/g, '')) || 0), 0))} outstanding</div></div>
   141→        <div className="bg-white rounded-xl border border-surface-dim p-3"><div className="text-[10px] text-on-surface-variant uppercase tracking-wider font-bold">Collected</div><div className="text-headline-sm mt-1 tabular-nums text-secondary">{money(items.reduce((s, x) => s + (parseFloat(x.paid.replace(/[−$,]/g, '')) || 0), 0))}</div><div className="text-[11px] text-on-surface-variant mt-0.5">across all statements</div></div>
   142→        <div className="bg-white rounded-xl border border-surface-dim p-3"><div className="text-[10px] text-on-surface-variant uppercase tracking-wider font-bold">Overdue</div><div className="text-headline-sm mt-1 tabular-nums text-error">{items.filter(s => s.status === 'Overdue').length}</div><div className="text-[11px] text-on-surface-variant mt-0.5">{money(items.filter(s => s.status === 'Overdue').reduce((s, x) => s + (parseFloat(x.dueAmt.replace(/[$,]/g, '')) || 0), 0))} past due</div></div>
   143→      </div>
   144→
   145→      {/* CREATE ACTIONS ROW */}
   146→      <div className="flex items-center gap-2 mb-3 flex-wrap">
   147→        <span className="text-label-sm text-on-surface-variant uppercase tracking-wider font-bold mr-1">Create actions:</span>
   148→        <button className="h-8 px-3 rounded-lg bg-white border border-surface-dim hover:bg-primary hover:text-white text-on-surface text-label-sm flex items-center gap-1.5" onClick={() => setDrawerId('__new__')}><span className="material-symbols-outlined text-[16px]">summarize</span>Statement</button>
   149→        <button className="h-8 px-3 rounded-lg bg-white border border-surface-dim hover:bg-primary hover:text-white text-on-surface text-label-sm flex items-center gap-1.5" onClick={() => setBatchOpen(true)}><span className="material-symbols-outlined text-[16px]">campaign</span>Batch send</button>
   150→        <button className="h-8 px-3 rounded-lg bg-white border border-surface-dim hover:bg-primary hover:text-white text-on-surface text-label-sm flex items-center gap-1.5" onClick={() => onToast('New customer — party drawer')}><span className="material-symbols-outlined text-[16px]">person_add</span>Customer</button>
   151→        <button className="h-8 px-3 rounded-lg bg-white border border-surface-dim hover:bg-primary hover:text-white text-on-surface text-label-sm flex items-center gap-1.5" onClick={() => onToast('New invoice')}><span className="material-symbols-outlined text-[16px]">receipt_long</span>Invoice</button>
   152→        <button className="h-8 px-3 rounded-lg bg-surface-container hover:bg-surface-container-high text-label-sm flex items-center gap-1" onClick={() => onToast('Global Create flyout')}><span className="material-symbols-outlined text-[16px]">apps</span>Show all</button>
   153→      </div>
   154→
   155→      {/* TABLE CARD */}
   156→      <div className="bg-white rounded-xl shadow-sm border border-surface-dim overflow-hidden">
   157→        <div className="flex items-center gap-1 px-4 h-11 border-b border-surface-dim overflow-x-auto">
   158→          {tabs.map(t => <button key={t} className={`h-full px-3 text-label-md tab-underline ${tab === t ? 'active' : 'text-on-surface-variant'}`} onClick={() => { setTab(t); setSelected(new Set()); }}>{t}</button>)}
   159→        </div>
   160→        <div className="px-4 py-2 flex items-center gap-2 border-b border-surface-dim flex-wrap">
   161→          <div className="relative flex items-center flex-1 max-w-sm min-w-[180px]"><span className="material-symbols-outlined absolute left-2 text-[15px] text-outline">search</span><input className="w-full h-7 pl-7 pr-2 bg-surface-container-low border border-surface-dim rounded text-body-sm focus:outline-none focus:border-primary" placeholder="Filter by customer, period…" value={q} onChange={e => setQ(e.target.value)} /></div>
   162→          <button className="h-7 px-2 border border-surface-dim rounded text-label-sm hover:bg-surface-container-low flex items-center gap-1"><span className="material-symbols-outlined text-[14px]">calendar_today</span>Date range</button>
   163→          <button className="h-7 px-2 border border-surface-dim rounded text-label-sm hover:bg-surface-container-low flex items-center gap-1" onClick={() => onToast('Saved views')}><span className="material-symbols-outlined text-[14px]">bookmark</span>Saved views</button>
   164→          <span className="ml-auto text-[11px] text-on-surface-variant">{filtered.length} of {items.length}</span>
   165→        </div>
   166→        {selected.size > 0 && (
   167→          <div className="bulk-bar show">
   168→            <span className="text-[12px] font-semibold">{selected.size} selected</span>
   169→            <span className="w-px h-4 bg-white/30" />
   170→            <button onClick={() => bulkAction('send')}><span className="material-symbols-outlined text-[14px] align-middle mr-1">send</span>Send</button>
   171→            <button onClick={() => bulkAction('print')}><span className="material-symbols-outlined text-[14px] align-middle mr-1">print</span>Print</button>
   172→            <button onClick={() => bulkAction('remind')}><span className="material-symbols-outlined text-[14px] align-middle mr-1">notifications_active</span>Remind</button>
   173→            <span className="w-px h-4 bg-white/30" />
   174→            <button onClick={() => bulkAction('delete')}><span className="material-symbols-outlined text-[14px] align-middle mr-1">delete</span>Delete</button>
   175→            <button className="ml-auto" onClick={() => bulkAction('close')}><span className="material-symbols-outlined text-[16px]">close</span></button>
   176→          </div>
   177→        )}
   178→        <table className="w-full text-left text-body-sm">
   179→          <thead className="bg-surface-container-low text-label-sm text-on-surface-variant uppercase tracking-wider">
   180→            <tr>
   181→              <th className="w-8 px-3 py-2"><input type="checkbox" className="row-check" checked={allChecked} ref={el => { if (el) el.indeterminate = someChecked && !allChecked; }} onChange={toggleAll} /></th>
   182→              <th className="px-3 py-2">Customer</th><th className="px-3 py-2">Period</th><th className="px-3 py-2">Sent</th><th className="px-3 py-2">Due</th><th className="px-3 py-2 text-right">Balance</th><th className="px-3 py-2">Status</th><th className="w-8" />
   183→            </tr>
   184→          </thead>
   185→          <tbody className="divide-y divide-surface-dim">
   186→            {filtered.map(s => (
   187→              <tr key={s.id} className="hover:bg-surface-container-low cursor-pointer" onClick={() => setDrawerId(s.id)}>
   188→                <td className="px-3 py-2"><input type="checkbox" className="row-check" checked={selected.has(s.id)} onClick={e => e.stopPropagation()} onChange={() => toggleRow(s.id)} /></td>
   189→                <td className="px-3 py-2"><div className="flex items-center gap-2"><div className={`st-avatar ${s.avatarColor}`}>{s.avatarLetter}</div><div className="min-w-0"><div className="font-medium truncate">{s.customer}</div><div className="text-[11px] text-on-surface-variant">{s.email}</div></div></div></td>
   190→                <td className="px-3 py-2 text-on-surface-variant">{s.period}</td>
   191→                <td className="px-3 py-2 text-on-surface-variant">{s.date === '2026-11-30' ? '—' : s.date}</td>
   192→                <td className={`px-3 py-2 ${s.status === 'Overdue' ? 'text-error' : 'text-on-surface-variant'}`}>{s.due}</td>
   193→                <td className={`px-3 py-2 text-right tabular-nums ${s.status === 'Overdue' ? 'text-error' : ''}`}>{s.balance}</td>
   194→                <td className="px-3 py-2"><span className={`st ${statusClass(s.status)}`}>{s.status}</span></td>
   195→                <td className="px-2 py-2 text-right relative">
   196→                  <button className="w-6 h-6 rounded hover:bg-surface-container flex items-center justify-center" onClick={e => { e.stopPropagation(); setRowMenuForId(rowMenuForId === s.id ? null : s.id); }}><span className="material-symbols-outlined text-[16px] text-on-surface-variant">more_horiz</span></button>
   197→                  {rowMenuForId === s.id && (
   198→                    <div className="row-menu open" style={{ position: 'absolute', right: 0, top: '100%', marginTop: 4 }} onClick={e => e.stopPropagation()}>
   199→                      <div className="menu-header">This statement</div>
   200→                      <button onClick={() => rowAction('view', s)}><span className="material-symbols-outlined">open_in_new</span>Open statement</button>
   201→                      <button onClick={() => rowAction('edit', s)}><span className="material-symbols-outlined">edit</span>Edit</button>
   202→                      <button onClick={() => rowAction('duplicate', s)}><span className="material-symbols-outlined">content_copy</span>Duplicate for next period</button>
   203→                      <div className="sep" />
   204→                      <div className="menu-header">Send</div>
   205→                      {s.status === 'Draft' && <button onClick={() => rowAction('send', s)}><span className="material-symbols-outlined">send</span>Send</button>}
   206→                      {(s.status === 'Sent' || s.status === 'Overdue' || s.status === 'Partial') && <button onClick={() => rowAction('remind', s)}><span className="material-symbols-outlined">notifications_active</span>Send reminder</button>}
   207→                      <button onClick={() => rowAction('print', s)}><span className="material-symbols-outlined">print</span>Print / download PDF</button>
   208→                      <div className="sep" />
   209→                      <div className="menu-header">Payment</div>
   210→                      {(s.status === 'Sent' || s.status === 'Overdue' || s.status === 'Partial') && <button onClick={() => rowAction('mark-paid', s)}><span className="material-symbols-outlined">check_circle</span>Mark as paid</button>}
   211→                      {(s.status === 'Sent' || s.status === 'Overdue' || s.status === 'Partial') && <button onClick={() => rowAction('receive-payment', s)}><span className="material-symbols-outlined">price_check</span>Receive payment…</button>}
   212→                      <div className="sep" />
   213→                      <button className="danger" onClick={() => rowAction('delete', s)}><span className="material-symbols-outlined">delete</span>Delete</button>
   214→                    </div>
   215→                  )}
   216→                </td>
   217→              </tr>
   218→            ))}
   219→          </tbody>
   220→        </table>
   221→      </div>
   222→
   223→      {/* New split menu */}
   224→      {newMenuOpen && (
   225→        <div className="row-menu open" style={{ position: 'absolute', right: 0, top: '100%', marginTop: 4 }} onClick={() => setNewMenuOpen(false)}>
   226→          <div className="menu-header">New statement</div>
   227→          <button onClick={() => setDrawerId('__new__')}><span className="material-symbols-outlined">person</span>Single customer</button>
   228→          <button onClick={() => setBatchOpen(true)}><span className="material-symbols-outlined">campaign</span>Batch send to many…</button>
   229→          <div className="sep" />
   230→          <button onClick={() => onToast('Generating from invoices — pick period')}><span className="material-symbols-outlined">receipt_long</span>Generate from invoices…</button>
   231→        </div>
   232→      )}
   233→
   234→      {/* Click-away for row menu */}
   235→      {rowMenuForId && <div style={{ position: 'fixed', inset: 0, zIndex: 150 }} onClick={() => setRowMenuForId(null)} />}
   236→
   237→      {/* Statement drawer */}
   238→      {drawerId && (
   239→        <>
   240→          <div className="drawer-scrim open" onClick={() => setDrawerId(null)} />
   241→          <div className="drawer open">
   242→            <div className="flex items-center justify-between px-5 h-14 border-b border-surface-dim shrink-0">
   243→              <div className="flex items-center gap-2 min-w-0">
   244→                <span className="material-symbols-outlined text-[20px] text-primary">summarize</span>
   245→                <div className="min-w-0">
   246→                  <div className="text-headline-sm truncate">{drawerId === '__new__' ? 'Statement ST-1103' : `Statement ${drawerId}`}</div>
   247→                  <div className="text-[11px] text-on-surface-variant">{drawerId === '__new__' ? 'New statement · Nov 2026' : `${drawerStmt?.customer || ''} · ${drawerStmt?.period || ''}`}</div>
   248→                </div>
   249→              </div>
   250→              <div className="flex items-center gap-1">
   251→                <button className="w-8 h-8 rounded hover:bg-surface-container flex items-center justify-center" title="Download"><span className="material-symbols-outlined text-[18px]">download</span></button>
   252→                <button className="w-8 h-8 rounded hover:bg-surface-container flex items-center justify-center" onClick={() => setDrawerId(null)}><span className="material-symbols-outlined text-[18px]">close</span></button>
   253→              </div>
   254→            </div>
   255→            <div className="flex-1 overflow-y-auto p-5 space-y-4">
   256→              <div className="grid grid-cols-2 gap-3">
   257→                <div><label className="field-label">Customer</label><input className="field-input" defaultValue={drawerStmt?.customer || ''} readOnly /></div>
