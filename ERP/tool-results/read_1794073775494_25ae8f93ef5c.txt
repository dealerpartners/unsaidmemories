   168→/* ─── The Document Engine Component ─── */
   169→interface DocumentEngineProps {
   170→  type?: DocumentType;
   171→  documentId?: string | null; // null = new document
   172→  onToast: (msg: string) => void;
   173→  onClose?: () => void;
   174→  partyId?: string | null; // prefill from envelope
   175→  /** Legacy controlled API (GlobalShell): open + activeTab + onSelectTab */
   176→  open?: boolean;
   177→  activeTab?: string;
   178→  onSelectTab?: (tab: string) => void;
   179→}
   180→
   181→export type DocTabType = 'edit' | 'email' | 'payor' | 'pdf';
   182→
   183→export default function DocumentEngine(props: DocumentEngineProps) {
   184→  // Legacy controlled mode (GlobalShell): open=false → render nothing.
   185→  if (props.open === false) return null;
   186→  return <DocumentEngineInner {...props} />;
   187→}
   188→
   189→function DocumentEngineInner({ type = 'Invoice', documentId, onToast, onClose, partyId: prefillPartyId, activeTab, onSelectTab }: DocumentEngineProps) {
   190→  const config = CONFIGS[type] || CONFIGS.Invoice;
   191→  const store = usePlatformStore();
   192→
   193→  const existing = documentId ? store.getDocument(documentId) : null;
   194→  const isNew = !existing;
   195→
   196→  // Form state
   197→  const [number, setNumber] = useState(existing?.number || `${config.numberPrefix}${String(store.documents.length + 1001)}`);
   198→  const [partyId, setPartyId] = useState(existing?.partyId || prefillPartyId || '');
   199→  const [status, setStatus] = useState(existing?.status || 'Draft');
   200→  const [date, setDate] = useState(existing?.date || new Date().toISOString().slice(0, 10));
   201→  const [dueDate, setDueDate] = useState(existing?.dueDate || '');
   202→  const [lines, setLines] = useState<DocumentLine[]>(existing?.lines || [{ id: 'L1', itemId: null, description: '', qty: 1, rate: 0, amount: 0, taxRate: 0, account: '', tags: [] }]);
   203→  const [shipping, setShipping] = useState(existing?.shipping || 0);
   204→  const [memo, setMemo] = useState(existing?.memo || '');
   205→  const [notes, setNotes] = useState(existing?.notes || '');
   206→  const [viewState, setViewState] = useState<'edit' | 'email' | 'payor' | 'pdf'>((activeTab as 'edit' | 'email' | 'payor' | 'pdf') || 'edit');
   207→  const [showPaymentModal, setShowPaymentModal] = useState(false);
   208→
   209→  // Legacy tab sync
   210→  const setView = (v: 'edit' | 'email' | 'payor' | 'pdf') => { setViewState(v); onSelectTab?.(v); };
   211→
   212→  const party = partyId ? store.getParty(partyId) : null;
   213→  
   214→  const subtotal = useMemo(() => lines.reduce((s, l) => s + l.amount, 0), [lines]);
   215→  const tax = useMemo(() => lines.reduce((s, l) => s + (l.amount * l.taxRate / 100), 0), [lines]);
   216→  const total = subtotal + tax + shipping;
   217→  const balance = existing ? existing.balance : total;
   218→
   219→  function updateLine(idx: number, updates: Partial<DocumentLine>) {
   220→    setLines(prev => prev.map((l, i) => {
   221→      if (i !== idx) return l;
   222→      const updated = { ...l, ...updates };
   223→      updated.amount = updated.qty * updated.rate;
   224→      return updated;
   225→    }));
   226→  }
   227→
   228→  function addLine() {
   229→    setLines(prev => [...prev, { id: `L${prev.length + 1}`, itemId: null, description: '', qty: 1, rate: 0, amount: 0, taxRate: 0, account: '', tags: [] }]);
   230→  }
   231→
   232→  function removeLine(idx: number) {
   233→    setLines(prev => prev.filter((_, i) => i !== idx));
   234→  }
   235→
   236→  function save() {
   237→    if (!partyId) { onToast(`${config.partyLabel} is required`); return; }
   238→    const docData = {
   239→      type, number, partyId, status, date, dueDate, lines, subtotal, tax: Math.round(tax * 100) / 100, shipping, total, amountPaid: existing?.amountPaid || 0, balance: total - (existing?.amountPaid || 0), memo, notes, tags: existing?.tags || [], attachments: existing?.attachments || [], orderId: existing?.orderId || null, poId: existing?.poId || null, channelId: existing?.channelId || null, createdBy: 'Current user',
   240→    };
   241→    if (existing) {
   242→      store.updateDocument(existing.id, docData);
   243→      store.recalcDocument(existing.id);
   244→      onToast(`${config.label} saved`);
   245→    } else {
   246→      const id = store.addDocument(docData as Omit<Document, 'id' | 'createdAt' | 'updatedAt' | 'activity'>);
   247→      onToast(`${config.label} ${number} created`);
   248→    }
   249→    onClose?.();
   250→  }
   251→
   252→  function send() {
   253→    if (!existing && isNew) { save(); return; }
   254→    if (existing) {
   255→      store.changeDocumentStatus(existing.id, 'Sent' as any);
   256→      onToast(`${config.label} ${number} sent to ${party?.email || 'customer'}`);
   257→    }
   258→  }
   259→
   260→  function recordPayment(amount: number, method: string, account: string) {
   261→    if (!existing) { onToast('Save the document first'); return; }
   262→    store.recordPayment(existing.id, amount, method, account);
   263→    onToast(`Payment of $${amount.toFixed(2)} recorded via ${method}`);
   264→    setShowPaymentModal(false);
   265→  }
   266→
   267→  return (
   268→    <div className="flex flex-col h-full">
   269→      {/* Header */}
   270→      <header className="h-12 bg-white border-b border-surface-dim flex items-center justify-between px-4 shrink-0">
   271→        <div className="flex items-center gap-1">
   272→          {['edit', 'email', 'payor', 'pdf'].map(v => (
   273→            <button key={v} className={`px-3 py-2 text-[13px] tab-underline ${viewState === v ? 'active' : 'text-on-surface-variant'}`} onClick={() => setView(v)}>
   274→              {v === 'edit' ? 'Edit' : v === 'email' ? 'Email view' : v === 'payor' ? 'Payor view' : 'PDF view'}
   275→            </button>
   276→          ))}
   277→        </div>
   278→        <div className="flex items-center gap-3 text-[13px] text-on-surface-variant">
   279→          {onClose && <button className="icon-btn" onClick={onClose}><span className="material-symbols-outlined">close</span></button>}
   280→        </div>
   281→      </header>
   282→
   283→      {/* Main */}
   284→      <div className="flex-1 flex overflow-hidden bg-[#faf9fc]">
   285→        {viewState === 'edit' && (
   286→          <div className="flex-1 overflow-y-auto p-6 flex justify-center">
   287→            <div className="w-full max-w-[900px] space-y-4">
   288→              {/* Document paper */}
   289→              <div className="bg-white border border-surface-dim shadow-sm rounded-sm min-h-[600px] p-8">
   290→                {/* Header row */}
   291→                <div className="flex justify-between items-start mb-6">
   292→                  <div>
   293→                    <h1 className="text-[24px] font-bold text-primary mb-1">{config.label.toUpperCase()}</h1>
   294→                    <div className="text-[13px] font-semibold">{number}</div>
   295→                  </div>
   296→                  <div className="text-right">
   297→                    <div className="text-[12px] text-on-surface-variant uppercase font-bold mb-1">{config.label} total</div>
   298→                    <div className="text-[24px] font-bold">${total.toFixed(2)}</div>
   299→                  </div>
   300→                </div>
   301→
   302→                {/* Party + dates */}
   303→                <div className="grid grid-cols-2 gap-8 mb-6">
   304→                  <div>
   305→                    <label className="block text-[11px] text-on-surface-variant uppercase font-bold mb-1">{config.partyLabel}</label>
   306→                    <select className="field-input" value={partyId} onChange={e => setPartyId(e.target.value)}>
   307→                      <option value="">Select {config.partyLabel.toLowerCase()}…</option>
   308→                      {store.getPartiesByRole(config.partyRole as any).map(p => (
   309→                        <option key={p.id} value={p.id}>{p.name}</option>
   310→                      ))}
   311→                    </select>
   312→                    {party && (
   313→                      <div className="mt-2 text-[12px] text-on-surface-variant">
   314→                        {party.address}, {party.city}, {party.state} {party.zip}
   315→                      </div>
   316→                    )}
   317→                  </div>
   318→                  <div className="space-y-3">
   319→                    <div>
   320→                      <label className="block text-[11px] text-on-surface-variant uppercase font-bold mb-1">{config.dateLabel}</label>
   321→                      <input type="date" className="field-input" value={date} onChange={e => setDate(e.target.value)} />
   322→                    </div>
   323→                    {config.dueLabel && (
   324→                      <div>
   325→                        <label className="block text-[11px] text-on-surface-variant uppercase font-bold mb-1">{config.dueLabel}</label>
   326→                        <input type="date" className="field-input" value={dueDate} onChange={e => setDueDate(e.target.value)} />
   327→                      </div>
   328→                    )}
   329→                  </div>
   330→                </div>
   331→
   332→                {/* Lines table */}
   333→                <table className="w-full text-left text-[13px] border border-surface-dim rounded">
   334→                  <thead>
   335→                    <tr className="text-[11px] text-on-surface-variant uppercase font-bold bg-surface-container-low border-b border-surface-dim">
   336→                      <th className="py-2 px-3 w-8"></th>
   337→                      {config.lineColumns.map(col => (
   338→                        <th key={col.key} className={`py-2 px-3 ${col.align === 'right' ? 'text-right' : ''}`} style={col.width ? { width: col.width } : undefined}>{col.label}</th>
   339→                      ))}
   340→                      <th className="py-2 px-3 w-8"></th>
   341→                    </tr>
   342→                  </thead>
   343→                  <tbody>
   344→                    {lines.map((line, idx) => (
   345→                      <tr key={idx} className="border-b border-surface-dim">
   346→                        <td className="py-2 px-3 text-on-surface-variant text-center"><span className="material-symbols-outlined text-[16px] cursor-grab">drag_indicator</span></td>
   347→                        <td className="py-2 px-3"><input className="w-full border-0 bg-transparent text-[13px] focus:outline-none focus:border-b focus:border-primary" placeholder="Description or item name" value={line.description} onChange={e => updateLine(idx, { description: e.target.value })} /></td>
   348→                        <td className="py-2 px-3"><input type="number" className="w-full border-0 bg-transparent text-[13px] text-right focus:outline-none focus:border-b focus:border-primary" value={line.qty} onChange={e => updateLine(idx, { qty: parseFloat(e.target.value) || 0 })} /></td>
   349→                        <td className="py-2 px-3"><input type="number" className="w-full border-0 bg-transparent text-[13px] text-right focus:outline-none focus:border-b focus:border-primary" value={line.rate} onChange={e => updateLine(idx, { rate: parseFloat(e.target.value) || 0 })} /></td>
   350→                        <td className="py-2 px-3 text-right tabular-nums">${line.amount.toFixed(2)}</td>
   351→                        <td className="py-2 px-3 text-center"><button className="text-on-surface-variant hover:text-error" onClick={() => removeLine(idx)}><span className="material-symbols-outlined text-[16px]">close</span></button></td>
   352→                      </tr>
   353→                    ))}
   354→                  </tbody>
   355→                </table>
   356→                <button className="mt-2 text-[12px] text-primary font-semibold hover:underline flex items-center gap-1" onClick={addLine}><span className="material-symbols-outlined text-[16px]">add</span>Add line</button>
   357→
   358→                {/* Totals */}
   359→                <div className="flex justify-end mt-6">
   360→                  <div className="w-64 space-y-2 text-[13px]">
   361→                    <div className="flex justify-between"><span className="text-on-surface-variant">Subtotal</span><span>${subtotal.toFixed(2)}</span></div>
   362→                    <div className="flex justify-between"><span className="text-on-surface-variant">Tax</span><span>${tax.toFixed(2)}</span></div>
   363→                    <div className="flex justify-between"><span className="text-on-surface-variant">Shipping</span><input type="number" className="w-20 border-0 bg-transparent text-right" value={shipping} onChange={e => setShipping(parseFloat(e.target.value) || 0)} /></div>
   364→                    <div className="flex justify-between font-bold border-t border-surface-dim pt-2"><span>Total</span><span>${total.toFixed(2)}</span></div>
   365→                    {existing && existing.amountPaid > 0 && (
   366→                      <>
   367→                        <div className="flex justify-between text-on-surface-variant"><span>Amount paid</span><span>${existing.amountPaid.toFixed(2)}</span></div>
   368→                        <div className="flex justify-between font-bold"><span>Balance due</span><span className={balance > 0 ? 'text-error' : 'text-secondary'}>${balance.toFixed(2)}</span></div>
   369→                      </>
   370→                    )}
   371→                  </div>
   372→                </div>
   373→
   374→                {/* Memo + notes */}
   375→                <div className="grid grid-cols-2 gap-6 mt-8">
   376→                  <div><label className="block text-[12px] text-on-surface-variant mb-1">Memo (on statement)</label><textarea className="field-input" style={{ height: 56 }} value={memo} onChange={e => setMemo(e.target.value)} /></div>
   377→                  <div><label className="block text-[12px] text-on-surface-variant mb-1">Note to {config.partyLabel.toLowerCase()}</label><textarea className="field-input" style={{ height: 56 }} value={notes} onChange={e => setNotes(e.target.value)} /></div>
   378→                </div>
   379→
   380→                {/* Activity log (existing docs only) */}
   381→                {existing && existing.activity.length > 0 && (
   382→                  <div className="mt-8 border-t border-surface-dim pt-4">
   383→                    <h3 className="text-[13px] font-bold mb-2">Activity</h3>
   384→                    {existing.activity.map((a, i) => (
